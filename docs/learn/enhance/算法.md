## 快排

```js
function quickSort(list) {
  // 当list.length <= 1时，退出递归
  if (list.length <= 1) return list;
  // 找到中间节点
  let mid = Math.floor(list.length / 2);
  // 以中间节点为基准点，比该节点大的值放到right数组中，否则放到left数组中
  let base = list.splice(mid, 1)[0];
  let left = [];
  let right = [];
  list.forEach((item) => {
    if (item > base) {
      right.push(item);
    } else {
      left.push(item);
    }
  });
  // 重新组合数组
  return quickSort(left).concat(base, quickSort(right));
}
```

## 二分法

对于升序，无重复元素的数组，查找指定元素

```js
var search = function (arr, target) {
  var mid;
  var left = 0;
  var right = arr.length - 1;
  while (left <= right) {
    mid = left + ((right - left) >> 1);
    if (arr[mid] > target) {
      right = mid - 1;
    } else if (arr[mid] < target) {
      left = mid + 1;
    } else {
      return mid;
    }
  }
  return -1;
};
```

## 买股票

题目： 给定一个整数数组，其中第 i 个元素代表了第 i 天的股票价格；
非负整数 fee 代表了交易股票的手续费用，求返回获得利润的最大值

输入： arr: [1, 12, 13, 9, 15, 8, 6, 16]； fee: 2

输出： 22

```js
function busyStock(list, fee) {
  let sum = 0;
  let price = list[0];
  let temp;
  for (let i = 1; i < list.length; i++) {
    if (list[i] < price) {
      price = list[i];
    } else {
      temp = list[i] - price - fee;
      if (temp) {
        sum += temp;
        price = list[i] - fee;
      }
    }
  }
  return sum;
}

busyStock([1, 12, 13, 9, 15, 8, 6, 16], 2);
```

## 斐波那契数列

```js
// 使用动态规划，将复杂的问题拆分，也就是：`F(N) = F(N - 1) + F(N - 2)`，用数组将已经计算过的值存起来
function fib(n) {
  if (n < 2) return n;
  let dp = [1, 1];
  for (let i = 2; i <= n; i++) {
    dp[i] = dp[i - 1] + dp[i - 2];
  }
  return dp[n - 1];
}
```

## 有效括号

描述：  
给定一个只包括 '('，')'，'{'，'}'，'['，']' 的字符串，判断字符串是否有效。

有效字符串需满足：

- 左括号必须用相同类型的右括号闭合。
- 左括号必须以正确的顺序闭合。
- 注意空字符串可被认为是有效字符串。

思路：  
将遇到的左侧括号依次存入，当遇到右侧括号时，取出最后存入的括号，进行匹配。

```js
function isValid(s) {
  if (s.length <= 1) return Boolean(!s.length);
  var temp = {
    "}": "{",
    "]": "[",
    ")": "(",
  };
  var result = [];
  var status = true;
  s.split("").forEach(function (item) {
    if (!temp[item]) {
      result.push(item);
    } else if (result.pop() !== temp[item]) {
      status = false;
    }
  });
  if (status) {
    status = !result.length;
  }
  return status;
}

console.log(isValid("[()]"));
```
